    <script>
      const form = document.getElementById("goal-form");
      const productField = document.getElementById("product-field");
      const productInput = document.getElementById("favorite-product");
      const feedback = document.getElementById("feedback");
      const maxGoals = 3;

      function updateProductField() {
        const selectedGoals = form.querySelectorAll('input[name="goal"]:checked');
        const showProduct = Array.from(selectedGoals).some(
          (goal) => goal.value === "jeść więcej wybranego produktu"
        );

        productField.classList.toggle("is-visible", showProduct);
        productField.setAttribute("aria-hidden", String(!showProduct));
        productInput.tabIndex = showProduct ? 0 : -1;
        productInput.required = showProduct;

        if (!showProduct) {
          productInput.value = "";
        }

        feedback.textContent = "";
        feedback.classList.remove("error");
      }

      form.addEventListener("change", (event) => {
        if (event.target.name === "goal") {
          const selectedGoals = form.querySelectorAll('input[name="goal"]:checked');
          if (event.target.checked && selectedGoals.length > maxGoals) {
            event.target.checked = false;
            updateProductField();
            feedback.textContent = "Możesz zaznaczyć maksymalnie 3 cele.";
            feedback.classList.add("error");
            return;
          }

          updateProductField();
        }
      });

      form.addEventListener("submit", (event) => {
        event.preventDefault();
        feedback.classList.remove("error");

        const selectedGoals = Array.from(
          form.querySelectorAll('input[name="goal"]:checked')
        );
        if (selectedGoals.length > maxGoals) {
          feedback.textContent = "Możesz zaznaczyć maksymalnie 3 cele.";
          feedback.classList.add("error");
          return;
        }

        if (selectedGoals.length === 0) {
          feedback.textContent = "Wybierz przynajmniej jeden cel, aby przejść dalej.";
          feedback.classList.add("error");
          form.querySelector('input[name="goal"]').focus();
          return;
        }

        if (
          selectedGoals.some(
            (goal) => goal.value === "jeść więcej wybranego produktu"
          ) &&
          !productInput.value.trim()
        ) {
          feedback.textContent = "Wpisz produkt, który chcesz częściej jeść.";
          feedback.classList.add("error");
          productInput.focus();
          return;
        }

        if (!form.reportValidity()) {
          return;
        }

        const goals = selectedGoals.map((goal) => goal.value);
        const product = productInput.value.trim();
        const productGoalIndex = goals.indexOf("jeść więcej wybranego produktu");
        if (productGoalIndex !== -1) {
          goals[productGoalIndex] += ` (${product})`;
        }
        feedback.textContent = `Świetnie! Twoje cele to: ${goals.join(", ")}.`;
      });
    </script>
